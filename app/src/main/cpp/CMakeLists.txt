cmake_minimum_required(VERSION 3.22)
project(avanevis_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory(whisper.cpp)
add_library(avanevis_whisper SHARED whisper_bridge.cpp)
target_include_directories(avanevis_whisper PRIVATE whisper.cpp/include)
target_link_libraries(avanevis_whisper PRIVATE whisper log)
